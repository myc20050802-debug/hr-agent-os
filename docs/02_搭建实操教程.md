## 第 0 步 · 决策：别急着写代码

### 0.1 先回答 5 个问题（写在纸上，不要跳过）

| 问题 | 为什么必须先答 | 示例答案 |
|---|---|---|
| 第一个种子客户是谁？ | 决定你对接哪个 ATS、用飞书还是钉钉 | 一家 600 人的互联网公司，用飞书 + Moka |
| 第一个 Agent 做哪个？ | 贪多必死 | 员工自助问答（风险最低） |
| 谁是使用者？谁是买单人？ | 使用者觉得好用 ≠ 买单人愿意付钱 | 使用者=HR专员，买单人=HRD |
| 数据在哪？拿得到吗？ | 拿不到数据，Agent 就是空壳 | 制度在飞书云文档，员工数据在 HRIS，需接口权限 |
| 怎么算成功？ | 没有指标就无法验收 | 试点 4 周内 HR 日均使用 ≥ 3 次 |

**产出**：一页纸《项目启动单》，包含上面 5 个答案 + 负责人 + 时间点。

### 0.2 技术选型（可直接照抄，也可替换同类）

| 模块 | 推荐 | 备选 | 选它的理由 |
|---|---|---|---|
| 前端 | React + TypeScript + Vite | Vue 3 | 后台表单/状态多，TS 能防住大量低级错误 |
| 后端 | Python + FastAPI | Java Spring Boot / Node NestJS | AI 生态最好，异步任务天然适配 |
| 数据库 | PostgreSQL 16 | MySQL 8 | 行级安全（RLS）天然支持多租户隔离 |
| 向量库 | pgvector 扩展 | Milvus / Qdrant | 一个库搞定，少一套运维 |
| 缓存/队列 | Redis + Celery | RabbitMQ | 任务重试、限流、异步调用 |
| 对象存储 | S3 兼容（MinIO 自建 / 云 OSS） | — | 存简历、合同、材料，必须加密 |
| 大模型 | 统一 LLM 网关（多厂商可切） | 直连某一家 | 客户一定会问「能不能换模型」 |
| 部署 | Docker Compose（起步）→ K8s（规模化） | — | 先简单，别一上来搞 K8s |

```sql
ALTER TABLE candidates ENABLE ROW LEVEL SECURITY;
CREATE POLICY tenant_isolation ON candidates
  USING (tenant_id = current_setting('app.tenant_id')::uuid);
```

```python
# backend/app/middleware/tenant.py —— 每个请求进来自动设置租户上下文
from fastapi import Request
from starlette.middleware.base import BaseHTTPMiddleware

class TenantMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        tenant_id = request.headers.get("X-Tenant-Id")   # 生产环境从 JWT 解析，不要信请求头
        async with request.app.state.db.session() as s:
            await s.execute("SELECT set_config('app.tenant_id', :t, true)", {"t": tenant_id})
        return await call_next(request)
```

> **为什么必须这么做**：AI 生成的 SQL 或检索可能忘记带租户过滤。数据库层兜底，是最后一道防线。

### 2.4 知识库入库（4 步走）

```python
# backend/app/services/kb_ingest.py
async def ingest_document(tenant_id: str, title: str, text: str, meta: dict):
    # 1. 切片：按语义切，不要按固定字数硬切
    chunks = semantic_split(text, target_tokens=400, overlap=60)
    # 2. 生成向量（用统一网关，方便换模型）
    vectors = await llm_gateway.embed(chunks)
    # 3. 入库
    for c, v in zip(chunks, vectors):
        await db.execute(insert(KbChunk).values(
            tenant_id=tenant_id, doc_id=doc_id, content=c, embedding=v, meta=meta))
    # 4. 版本化：同一份制度新版本入库时，旧版本标记 expired，不物理删除（审计需要）
    await db.execute(update(KbDocument).where(...).values(status="expired"))
```

**切片三条经验**：
1. 按「标题层级 + 段落」切，保留章节路径写进 `meta`，回答时能说出「根据《员工手册》第 3 章」
2. 每条 chunk 前缀补上文档标题（提升检索命中率）
3. 制度类文档必须带 `effective_at`，过期文档自动降权

**✅ 验收卡**：
- `SELECT count(*) FROM candidates;` 有数据
- 用一句「年假怎么算」检索，Top3 命中相关章节
- 用 A 租户身份查 B 租户数据，返回 0 行

### 3.2 任务规划拆解（Planner）

```python
# agents/planner.py
PLANNER_PROMPT = """你是HR Agent的规划器。用户目标：{goal}
可用工具：{tools_json}
要求：
1. 输出 JSON：{"steps":[{"id":1,"intent":"...","tool":"...","params":{...},"depends_on":[0],"high_risk":false}]}
2. 只允许使用列表中的工具；参数必须符合 Schema。
3. 涉及删除/发送/金额/对外沟通的步骤，high_risk 必须为 true。
4. 步骤数不超过 8 步。无法完成时输出 {"error":"原因"}。"""

async def plan(task, tools) -> dict:
    raw = await llm_gateway.chat_json(PLANNER_PROMPT.format(
        goal=task.goal, tools_json=tools.schema_dump()))
    plan = validate_plan(raw, tools)          # 校验：工具存在？参数合法？步骤数？
    await db.execute(insert(AgentTask).values(**{
        "id": task.id, "goal": task.goal, "plan": plan["steps"], "status": "running"}))
    return plan
```

**三条硬规则**：
1. **计划必须落库**（上面已落），出问题能回放
2. **用 JSON Schema 强约束输出**，解析失败就重试一次，再失败转人工——不要让模型自由发挥
3. **计划要给人看**：UI 上先展示「我打算这么干」，人点确认后再跑。这是信任的关键设计

### 3.3 工具注册中心

```python
# tools/registry.py
@dataclass
class ToolSpec:
    name: str
    desc: str                      # 给模型看的自然语言说明，写得越清楚，模型调得越准
    params_schema: dict            # JSON Schema
    handler: Callable
    high_risk: bool = False
    idempotency_key: str | None = None   # 模板，如 "invite:{candidate_id}:{round}"
    timeout_s: int = 15

REGISTRY: dict[str, ToolSpec] = {}

def tool(**kw):
    def deco(fn):
        spec = ToolSpec(handler=fn, **kw)
        REGISTRY[kw["name"]] = spec
        return fn
    return deco

# ---- 只读工具：免审 ----
@tool(name="hr.get_leave_balance", desc="查询指定员工的年假/调休余额。只能查发起人本人，除非发起人是HR。",
      params_schema={"type":"object","properties":{"user_id":{"type":"string"}},"required":["user_id"]})
async def get_leave_balance(ctx, user_id: str):
    assert_same_person_or_hr(ctx, user_id)          # 权限校验内置在工具里，不靠模型自觉
    return await hr_db.query_one("SELECT annual_leave_days FROM leave_balance WHERE user_id=%s", user_id)

# ---- 写工具：中等风险，执行前确认 ----
@tool(name="ats.update_stage", desc="更新候选人招聘阶段",
      params_schema={"type":"object","properties":{"candidate_id":{"type":"string"},"stage":{"type":"string"}},
                     "required":["candidate_id","stage"]},
      high_risk=False, idempotency_key="stage:{candidate_id}:{stage}")
async def update_stage(ctx, candidate_id: str, stage: str):
    return await ats_client.patch(f"/candidates/{candidate_id}", {"stage": stage})

# ---- 高风险工具：必须人工审核 ----
@tool(name="mail.send_offer", desc="向候选人发送Offer邮件。高风险操作，必须经人工审核。",
      params_schema={"type":"object","properties":{"candidate_id":{"type":"string"},"content":{"type":"string"}},
                     "required":["candidate_id","content"]},
      high_risk=True, idempotency_key="offer:{candidate_id}")
async def send_offer(ctx, candidate_id: str, content: str):
    if not ctx.approved:                      # 未过闸门，直接拒绝
        raise PermissionError("HIGH_RISK_REQUIRES_APPROVAL")
    return await mail.send(to=candidate_id, body=content)
```

### 3.4 执行器：重试 + 幂等（别让重试演变成重复发信）

```python
# agents/executor.py
from tenacity import retry, stop_after_attempt, wait_exponential

@retry(stop=stop_after_attempt(3), wait=wait_exponential(multiplier=1, max=10), reraise=True)
async def call_tool(task_id, step, ctx):
    spec = REGISTRY[step["tool"]]
    validate_params(spec, step["params"])                     # ① Schema 校验
    await guard.check_permission(ctx, spec)                   # ② 权限：以发起人身份
    safe_params = guard.mask_pii(step["params"])              # ③ 脱敏后再传给模型/外部
    if spec.high_risk:
        await guard.enqueue_approval(task_id, spec, safe_params)   # ④ 进闸门，抛出让流程暂停
        raise ApprovalRequired()
    idem = render_idem_key(spec.idempotency_key, step["params"])
    if idem and await redis.get(f"idem:{idem}"):              # ⑤ 幂等命中，不重复执行
        return await db.fetch_tool_call_result(task_id, step["id"])
    t0 = time.time()
    try:
        res = await asyncio.wait_for(spec.handler(ctx, **step["params"]), timeout=spec.timeout_s)
        if idem: await redis.setex(f"idem:{idem}", 86400, "1")
        await log_tool_call(task_id, spec, step, res, "ok", int((time.time()-t0)*1000))
        return res
    except Exception as e:
        await log_tool_call(task_id, spec, step, None, "error", int((time.time()-t0)*1000), str(e))
        raise
```

### 3.5 脱敏网关（三道关，必须在 Prompt 之前）

```python
# tools/guard.py
PII_PATTERNS = {
    "phone": r"1[3-9]\d{9}",
    "idcard": r"\d{17}[\dXx]",
    "bankcard": r"\d{16,19}",
}

def mask_pii(payload: dict) -> dict:
    """① 入模型前：把敏感值替换为占位符，模型永远看不到原文"""
    s = json.dumps(payload, ensure_ascii=False)
    for k, p in PII_PATTERNS.items():
        s = re.sub(p, f"[{k.upper()}]", s)
    return json.loads(s)

def scan_output(text: str) -> str:
    """② 模型输出后：兜底扫一遍，防止它把记忆里的原文吐出来"""
    for k, p in PII_PATTERNS.items():
        if re.search(p, text):
            audit_log("PII_LEAK_BLOCKED", text[:200])
            raise ValueError("OUTPUT_CONTAINS_PII")
    return text

DROP_FOR_MODEL = ["gender", "birth_date", "photo", "marital_status", "hukou"]
def strip_protected(fields: dict) -> dict:
    """③ 反歧视：物理删除受保护字段，而不是提示模型'别考虑性别'"""
    return {k: v for k, v in fields.items() if k not in DROP_FOR_MODEL}
```

> **记住这句话**：提示词会失效，代码不会。**凡是靠 Prompt 实现的合规，都要用代码再兜一层。**

```python
# agents/selfservice.py
async def answer(question: str, user: UserCtx):
    route = classify(question)     # 关键一步：先判断问题类型
    if route == "personal_data":   # 「我还有几天年假」

        return compose_with_data(question, data, source="HR系统实时数据")
    if route == "policy":          # 「年假怎么算」
        hits = await kb_search(question, tenant=user.tenant, top_k=5,
                               filter={"visibility": user.visible_scopes})
        if not hits or hits[0].score < 0.55:          # 低于阈值不硬答
            return escalate_to_hr(question, reason="知识库无匹配")
```

**必做四件事**：

## 附录 · 常见坑速查表

| 坑 | 症状 | 解法 |
|---|---|---|
| 一上来做 Multi-Agent | 效果不可控、调试困难、成本高 | 先用「单 Agent + 固定工具集」，真需要再拆 |
| 什么都用 AI | 提醒类功能又慢又贵还容易错 | 规则/定时任务优先，AI 只做判断类 |
| Prompt 里写业务规则 | 改一次要重新测一遍，无法审计 | 规则写进代码，Prompt 只留判断逻辑 |
| 记忆覆盖事实 | AI 说「人在职」，系统显示已离职 | 事实永远以系统 of record 为准 |
| 重试导致重复发送 | 候选人收到 2 封邀约 | 幂等键 + Redis 去重 |
| 用提示词做合规 | 偶尔还是泄漏敏感信息 | 代码层物理删字段 + 输出扫描 |
| 没有评测就上线 | 不知道改完是变好还是变坏 | 先写 50 条评测用例再写功能 |
| 全自动发 Offer | 一次事故职业生涯结束 | 高风险永远 L1/L2，人工闸门不可绕过 |
| 忽略成本 | 一个月 token 账单超过营收 | 模型路由 + 上下文裁剪 + 预算上限 |

---

> **最后一句实话**：这套平台的难点不在「接大模型」，而在**数据是否干净、权限是否可信、人愿不愿意用**。把这三件事做好，AI 是水到渠成的最后一层；这三件事做不好，接什么模型都白搭。
